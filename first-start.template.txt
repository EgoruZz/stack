# Заполни поля ниже и запусти:  bash ~/stack/apply-first-start.sh
# Скрипт проверит заполнение и разложит данные по личным файлам
# (~/.stack-secrets.sh и др.). Сам файл формы можно удалить после.
#
# ── ISP-прокси (proxy.market): host:port:login:password, срок подписки ──
PROXY_HOST=
PROXY_USER=
PROXY_PASS=
PROXY_EXPIRES=YYYY-MM-DD HH:MM +ZZZZ             # формат ГГГГ-ММ-ДД ЧЧ:ММ ±ЧЧММ; кавычки не обязательны
#
# ── Wolfram ID (для неинтерактивной активации лицензии; опционально) ──
WOLFRAM_ID=
WOLFRAM_PASS=
#
# ── git identity (для коммитов и PR) ──
GIT_NAME=
GIT_EMAIL=
#
# ── Sudo без пароля для агента (опционально) ──
# AGENT_SUDO=full    → NOPASSWD: ALL (удобно; ВСЁ, что запускается от тебя, получит root)
# AGENT_SUDO=none    → без изменений (пароль запрашивается вручную; обновления CLT — сам)
# Значение AGENT_SUDO=none/пусто = безопасный дефолт.
AGENT_SUDO=none
